## 2. Standard-Richtlinie auf DROP, dann erlauben (Server-Namespace)
$ on srv iptables -P INPUT DROP
$ on srv iptables -A INPUT -i lo -j ACCEPT
$ on srv iptables -A INPUT -m conntrack --ctstate ESTABLISHED,RELATED -j ACCEPT
$ on srv iptables -A INPUT -p tcp --dport 80 -j ACCEPT
$ on srv iptables -A INPUT -p icmp --icmp-type echo-request -m limit --limit 5/second -j ACCEPT
http 10.0.2.2:80: OK
http 10.0.2.2:8080: BLOCKIERT/keine Antwort
ping 10.0.2.2: OK

## 4. Regel einfügen (-I), ersetzen (-R) und löschen (-D)
$ on srv iptables -I INPUT 3 -p tcp --dport 8080 -s 10.0.1.0/24 -j ACCEPT
http 10.0.2.2:8080: OK
$ on srv iptables -R INPUT 3 -p tcp --dport 8080 -s 10.0.9.0/24 -j ACCEPT
http 10.0.2.2:8080: BLOCKIERT/keine Antwort
$ on srv iptables -D INPUT 3
$ on srv iptables -S INPUT
-P INPUT DROP
-A INPUT -i lo -j ACCEPT
-A INPUT -m conntrack --ctstate RELATED,ESTABLISHED -j ACCEPT
-A INPUT -p tcp -m tcp --dport 80 -j ACCEPT
-A INPUT -p icmp -m icmp --icmp-type 8 -m limit --limit 5/sec -j ACCEPT

## 5. NAT-Tabelle und forward in der Firewall mit iptables
$ on fw iptables -t nat -A POSTROUTING -o v-fw2 -j MASQUERADE
$ on fw iptables -P FORWARD DROP
$ on fw iptables -A FORWARD -m conntrack --ctstate ESTABLISHED,RELATED -j ACCEPT
$ on fw iptables -A FORWARD -i v-fw1 -o v-fw2 -p tcp --dport 80 -j ACCEPT
http 10.0.2.2:80: OK
http 10.0.2.2:8080: BLOCKIERT/keine Antwort

## 6. Dieselben Regeln aus Sicht von nft: iptables-nft legt Tabellen im nf_tables-Kern an
# Warning: table ip nat is managed by iptables-nft, do not touch!
# Warning: table ip filter is managed by iptables-nft, do not touch!
table ip nat {
	chain POSTROUTING {
		type nat hook postrouting priority srcnat; policy accept;
		oifname "v-fw2" counter packets 1 bytes 60 masquerade
	}
}
table ip filter {
	chain FORWARD {
		type filter hook forward priority filter; policy drop;
		ct state related,established counter packets 12 bytes 906 accept
		iifname "v-fw1" oifname "v-fw2" tcp dport 80 counter packets 1 bytes 60 accept
	}
}
